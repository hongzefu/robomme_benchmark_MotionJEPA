#!/bin/bash
# 维护计划第三步对拍单遍（计算节点内，经 launch.sh 的 srun 进入）：pass.sh <new|old> <遍名> <V9|X0> <identities 文件> 
#   new：改后代码 = robomme_benchmark-maint 检出（钉 ANCHOR_NEW），解释器借 noise 克隆 .venv，--src-root 指向 maint，带 --expect-ref 边生成边判定
#   old：旧代码 = robomme_benchmark-noise（f8f76fba）原样，不带 --expect-ref（旧代码没有该参数）
# 一律经各自代码里的 noise_run_gl.sh：输出根必须不存在、预算登记、来源报告、EXIT_CODE= 尾行。
set -u
N=/nfs/turbo/coe-chaijy-unreplicated/hongzefu
M=$N/maint-regress
NOISE=$N/robomme_benchmark-noise
MAINT=$N/robomme_benchmark-maint
export PY=$NOISE/.venv/bin/python
SIDE=$1; PASS=$2; SET=$3; IDS=$4
case "$SIDE" in
  new) C=$MAINT; EXPECT=$(cat $M/inputs/ANCHOR_NEW); export PYTHONPATH=$MAINT/src; EXTRA=(--expect-ref $MAINT/scripts/configs/noise-ref-20261003.json);;
  old) C=$NOISE; EXPECT=f8f76fbabc27d0b95ea9acbd57288e49411c7fb4; unset PYTHONPATH; EXTRA=();;
  *) echo "未知 side $SIDE"; echo "EXIT_CODE=2"; exit 2;;
esac
[ "$(git -C $C rev-parse HEAD)" = "$EXPECT" ] || { echo "错误: $C HEAD 不是 $EXPECT"; echo "EXIT_CODE=2"; exit 2; }
[ -z "$(git -C $C status --short --ignore-submodules=dirty)" ] || { echo "错误: $C 工作区不干净"; echo "EXIT_CODE=2"; exit 2; }
if [ "$SIDE" = new ]; then
  got=$($PY -c "import robomme_hard;print(robomme_hard.__file__)" 2>/dev/null)
  case "$got" in $MAINT/src/*) echo "IMPORT_CHECK=PASS $got";; *) echo "IMPORT_CHECK=FAIL $got"; echo "EXIT_CODE=2"; exit 2;; esac
fi
for f in /usr/share/vulkan/icd.d/nvidia_icd.x86_64.json /usr/share/vulkan/icd.d/nvidia_icd.json; do [ -f "$f" ] && { export VK_ICD_FILENAMES=$f; break; }; done
nvidia-smi --query-gpu=name,driver_version --format=csv,noheader | sed 's/^/GPU_DRIVER /'
COMMON=(--pass "$PASS" --log "$M/logs/$PASS.log" --provenance-out "$M/prov/$PASS.json"
        --budget-ledger "$M/budget-ledger.jsonl" --budget-caps "$M/inputs/budget-caps.json")
OUT=/tmp/mreg-$PASS; STAGE=$M/gen/$PASS
[ ! -e "$OUT" ] || { echo "错误: 节点输出 $OUT 已存在"; echo "EXIT_CODE=4"; exit 4; }
if [ "$SET" = V9 ]; then
  REAL=($PY $C/scripts/parity/hard_parity.py generate --side H2 --tier v9 --manifest $M/inputs/delivery.local.json
        --identities "$IDS" --specs-root $M/inputs/specs-root --src-root $C --workers 4 --gpu 0 --out $OUT --stage $STAGE "${EXTRA[@]}")
elif [ "$SET" = X0 ]; then
  export ROBOMME_HARD_XHARD0_IN_TEST_HARD=1
  REAL=($PY $C/scripts/parity/hard_parity.py generate --side H --tier xhard0 --manifest $M/inputs/xhard0_manifest.json
        --identities "$IDS" --src-root $C --workers 4 --gpu 0 --out $OUT --stage $STAGE "${EXTRA[@]}")
else echo "未知集合 $SET"; echo "EXIT_CODE=2"; exit 2; fi
n=$(grep -c . "$IDS")
cd $C && bash $C/scripts/parity/noise_run_gl.sh "${COMMON[@]}" --kind gen --out-root "$STAGE" \
  --attempts $n --resets $((n*3)) --retries 0 -- "${REAL[@]}"
rc=$?
# 收尾：把节点上未搬运的局 ship 并 finalize（辅助步骤，同一节点进程内）
$PY $C/scripts/parity/noise_run.py ship --src "$OUT" --stage "$STAGE" --finalize; rc2=$?
rm -rf "$OUT"
echo "MREG_PASS_END side=$SIDE pass=$PASS set=$SET rc=$rc ship_rc=$rc2 stage=$STAGE"
[ $rc -eq 0 ] && [ $rc2 -eq 0 ]; exit $?
